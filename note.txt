## Boot health (must be Partition or apply/bootstrap die on reboot)
# ./out/bin/pertiskctl -e <IP>:50000 logs pertiskd -n 40 | rg 'STATE volume|EPHEMERAL /var'
# expect: source=Partition , ephemeral_mounted=true
# if EphemeralTmpfs / no EPHEMERAL → disk modules (sd_mod+t10-pi, ext4) not loaded; rebuild image

## Bootstrap CP
# Discover CP IP (scan :50000 or Serial), then:
./out/bin/pertiskctl gen config lab-ha https://<CP_IP>:6443 -o ./out/cluster
./out/bin/pertiskctl -e <CP_IP>:50000 apply -f ./out/cluster/controlplane.yaml
./out/bin/pertiskctl -e <CP_IP>:50000 bootstrap
./out/bin/pertiskctl -e <CP_IP>:50000 kubeconfig -f ./out/cluster/admin.conf
./out/bin/pertiskctl -e <CP_IP>:50000 join-config -f ./out/cluster/worker.yaml

## Join workers (REQUIRED — bootstrap does NOT apply these to the API)
# 1) Bootstrap token Secret from worker.yaml cluster.token (id = part before '.')
#    name: bootstrap-token-<id>  ns: kube-system  type: bootstrap.kubernetes.io/token
#    stringData: token-id, token-secret, usage-bootstrap-authentication/signing=true,
#                auth-extra-groups: system:bootstrappers:kubeadm:default-node-token
# 2) kubectl --kubeconfig ./out/cluster/admin.conf apply -f examples/bootstrap/node-rbac.yaml
# 3) kubectl --kubeconfig ./out/cluster/admin.conf label node lab-ha-cp-1 node-role.kubernetes.io/control-plane=
# 4) Unique hostname per worker, then apply:
#    ./out/bin/pertiskctl -e <WK_IP>:50000 apply -f ./out/cluster/worker.yaml
# 5) If CSR Unauthorized / system:anonymous → token/RBAC missing or kubelet hit backoff;
#    apply Secret+RBAC, then re-apply worker.yaml to restart kubelet
# 6) CNI: kubectl --kubeconfig ./out/cluster/admin.conf apply -f examples/cni/kube-flannel.yaml

kubectl --kubeconfig ./out/cluster/admin.conf get nodes -o wide

## Pitfalls
# - After Proxmox/host reboot: without STATE Partition + EPHEMERAL /var, cluster is gone (RAM-only).
# - kubeconfig "admin kubeconfig missing; bootstrap first" → not bootstrapped on disk STATE.
# - Workers join fail with Unauthorized → missing bootstrap-token Secret and/or node-rbac.yaml.
# - Same hostname on two workers → node identity clash; always unique hostnames.
